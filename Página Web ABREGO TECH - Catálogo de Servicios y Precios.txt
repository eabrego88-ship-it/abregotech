    // Variables para gestión de estado del modal


    let selectedService = '';


    let selectedPrice = '';


    const phoneNumber = "50760000000"; // Reemplazar con el número real de ABREGO TECH


    function openModal(serviceName, price) {


      selectedService = serviceName;


      selectedPrice = price;


      document.getElementById('modalServiceName').innerText = serviceName;


      document.getElementById('modalServicePrice').innerText = 'B/. ' + price;


      document.getElementById('whatsappModal').classList.add('active');


    }


    function closeModal() {


      document.getElementById('whatsappModal').classList.remove('active');


    }


    function sendToWhatsApp() {


      const message = `Hola, ABREGO TECH. Estoy interesado en el servicio: ${selectedService} (Precio ref: B/. ${selectedPrice}). Quisiera solicitar más información o agendar una cita.`;


      const encodedMessage = encodeURIComponent(message);


      const whatsappUrl = `https://wa.me/${phoneNumber}?text=${encodedMessage}`;


      window.open(whatsappUrl, '_blank');


      closeModal();


    }


    // Buscador interactivo


    function filterServices() {


      const input = document.getElementById('searchInput').value.toLowerCase();


      const rows = document.querySelectorAll('.service-row');


      const blocks = document.querySelectorAll('.category-block');


      rows.forEach(row => {


        const text = row.innerText.toLowerCase();


        if (text.includes(input)) {


          row.classList.remove('hidden');


        } else {


          row.classList.add('hidden');


        }


      });


      // Ocultar categorías vacías en búsqueda


      blocks.forEach(block => {


        const visibleRows = block.querySelectorAll('.service-row:not(.hidden)');


        if (visibleRows.length === 0) {


          block.classList.add('hidden');


        } else {


          block.classList.remove('hidden');


        }


      });


    }


    // Filtro por etiquetas de categoría


    function filterCategory(categoryClass, element) {


      // Activar tag visualmente


      document.querySelectorAll('.tag').forEach(tag => tag.classList.remove('active'));


      element.classList.add('active');


      // Limpiar buscador de texto


      document.getElementById('searchInput').value = '';


      const blocks = document.querySelectorAll('.category-block');


      const rows = document.querySelectorAll('.service-row');


      // Mostrar todas las filas primero


      rows.forEach(row => row.classList.remove('hidden'));


      if (categoryClass === 'all') {


        blocks.forEach(block => block.classList.remove('hidden'));


      } else {


        blocks.forEach(block => {


          if (block.getAttribute('data-category') === categoryClass) {


            block.classList.remove('hidden');


          } else {


            block.classList.add('hidden');


          }


        });


      }


    }
